name: Kalshi Bot - Consolidated Paper Monitor
on:
  workflow_dispatch:
  schedule:
    - cron: '*/5 * * * *'
permissions:
  contents: write
concurrency:
  group: kalshi-paper-state
  cancel-in-progress: false
jobs:
  monitor:
    runs-on: ubuntu-24.04
    timeout-minutes: 6
    steps:
      - name: Download bot code
        uses: actions/checkout@v4
        with:
          persist-credentials: true
      - name: Install Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Install dependencies
        run: python -m pip install -r requirements.txt
      - name: Test trading rules
        run: python -m pytest -q
      - name: Run consolidated read-only monitor
        run: python -u paper_runner.py --duration 270
